option(HOS_BUILD_BENCHMARK "Build matched FFTW/OpenMP and CUDA benchmarks" OFF)
if(HOS_BUILD_BENCHMARK)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  find_package(OpenMP REQUIRED)
  execute_process(COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/generate_cpu_backend.py
    ${CMAKE_CURRENT_BINARY_DIR}/cpu_generated COMMAND_ERROR_IS_FATAL ANY)
  configure_file(benchmark.cu benchmark_cpu.cpp COPYONLY)
  add_executable(hos_benchmark_cpu ${CMAKE_CURRENT_BINARY_DIR}/benchmark_cpu.cpp)
  target_compile_features(hos_benchmark_cpu PRIVATE cxx_std_17)
  target_compile_definitions(hos_benchmark_cpu PRIVATE HOS_CPU)
  target_include_directories(hos_benchmark_cpu PRIVATE ${CMAKE_CURRENT_SOURCE_DIR} ${CMAKE_CURRENT_BINARY_DIR})
  target_link_libraries(hos_benchmark_cpu PRIVATE OpenMP::OpenMP_CXX fftw3_threads fftw3f_threads fftw3 fftw3f)
  add_executable(hos_benchmark_gpu benchmark.cu)
  target_compile_features(hos_benchmark_gpu PRIVATE cxx_std_17)
  set_target_properties(hos_benchmark_gpu PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED ON)
  target_link_libraries(hos_benchmark_gpu PRIVATE CUDA::cudart CUDA::cufft fftw3)
  add_executable(hos_compare compare_benchmark.cpp)
  target_compile_features(hos_compare PRIVATE cxx_std_17)
endif()
